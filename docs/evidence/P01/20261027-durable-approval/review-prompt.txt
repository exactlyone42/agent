use crate::{
    store::{Claim, StoreError, TaskStore},
    task::CancellationToken,
};
use rig_agent::agent::{
    AgentHook, DispatchAction, DispatchEvent, HookContext, OutcomeAction, OutcomeEvent,
};
use std::{
    sync::{
        Arc, Mutex,
        atomic::{AtomicBool, Ordering},
    },
    time::Duration,
};

#[derive(Clone)]
pub struct ApprovalGate {
    store: Arc<Mutex<TaskStore>>,
    task_id: String,
    cancellation: CancellationToken,
    manual: bool,
    denied: Arc<AtomicBool>,
}

impl ApprovalGate {
    pub fn new(
        store: Arc<Mutex<TaskStore>>,
        task_id: String,
        cancellation: CancellationToken,
        manual: bool,
    ) -> Self {
        Self {
            store,
            task_id,
            cancellation,
            manual,
            denied: Arc::new(AtomicBool::new(false)),
        }
    }

    pub fn was_denied(&self) -> bool {
        self.denied.load(Ordering::SeqCst)
    }
}

impl AgentHook for ApprovalGate {
    async fn on_dispatch(&self, _: &HookContext, event: DispatchEvent<'_>) -> DispatchAction {
        let Some(name) = event.tool_name() else {
            return DispatchAction::proceed();
        };
        if self.cancellation.is_cancelled() {
            return DispatchAction::stop("任務已要求停止");
        }
        let (Some(call_id), Some(args)) = (event.call_id, event.tool_args()) else {
            return DispatchAction::stop("工具缺少識別或參數，不能執行");
        };
        let prepared = match self.store.lock() {
            Ok(mut store) => store.prepare_intent(&self.task_id, &call_id.to_string(), name, args),
            Err(_) => return DispatchAction::stop("任務紀錄狀態異常，不能執行工具"),
        };
        let prepared = match prepared {
            Ok(prepared) => prepared,
            Err(StoreError::InvalidSpec) => {
                return DispatchAction::skip("工具參數或授權範圍不符合限制");
            }
            Err(_) => return DispatchAction::stop("無法保存工具意圖，未執行工具"),
        };
        loop {
            if self.cancellation.is_cancelled() {
                return DispatchAction::stop("任務已要求停止");
            }
            let claim = match self.store.lock() {
                Ok(mut store) => store.claim(&prepared),
                Err(_) => return DispatchAction::stop("任務紀錄狀態異常，不能執行工具"),
            };
            match claim {
                Ok(Claim::Granted) => return DispatchAction::proceed(),
                Ok(Claim::Denied) if self.manual => {
                    self.denied.store(true, Ordering::SeqCst);
                    return DispatchAction::stop("工具未獲批准或批准已失效");
                }
                Ok(Claim::Denied) => return DispatchAction::skip("使用者未授權此工具操作"),
                Err(_) => return DispatchAction::stop("無法領用批准，未執行工具"),
                Ok(Claim::Pending) => {}
            }
            tokio::select! {
                _ = self.cancellation.cancelled() => return DispatchAction::stop("任務已要求停止"),
                _ = tokio::time::sleep(Duration::from_millis(200)) => {}
            }
        }
    }

    async fn on_outcome(&self, _: &HookContext, event: OutcomeEvent<'_>) -> OutcomeAction {
        let Some(result) = event.tool_result() else {
            return OutcomeAction::proceed();
        };
        if result.is_skipped() {
            return OutcomeAction::proceed();
        }
        let Some(call_id) = event.call_id else {
            return OutcomeAction::stop("工具結果缺少識別，不能確認成果");
        };
        let saved = match self.store.lock() {
            Ok(mut store) => {
                store.receipt(&self.task_id, &call_id.to_string(), result.is_success())
            }
            Err(_) => return OutcomeAction::stop("任務紀錄狀態異常，請核對已執行的動作"),
        };
        if saved.is_err() {
            return OutcomeAction::stop("無法保存工具結果，請核對已執行的動作");
        }
        OutcomeAction::proceed()
    }
}

fn append(
    transaction: &Transaction<'_>,
    task_id: &str,
    kind: &str,
    state: Option<&str>,
    related_id: &str,
) -> Result<(), StoreError> {
    let seq = transaction.query_row(
        "UPDATE tasks SET last_seq=last_seq+1, recorded_state=coalesce(?2,recorded_state), updated_at=strftime('%Y-%m-%dT%H:%M:%fZ','now')
         WHERE task_id=?1 AND recorded_state IN ('running','waiting_approval') RETURNING last_seq", params![task_id,state], |r| r.get::<_, i64>(0))
        .optional()?.ok_or(StoreError::Conflict)?;
    transaction.execute(
        "INSERT INTO events(event_id,task_id,seq,event_type,related_id) VALUES (?1,?2,?3,?4,?5)",
        params![Uuid::new_v4().to_string(), task_id, seq, kind, related_id],
    )?;
    Ok(())
}

impl TaskStore {
    pub fn prepare_intent(
        &mut self,
        task_id: &str,
        call_id: &str,
        tool_name: &str,
        args: &str,
    ) -> Result<PreparedIntent, StoreError> {
        if call_id.is_empty() || call_id.len() > 256 || args.len() > 6 * FILE_LIMIT as usize + 1024
        {
            return Err(StoreError::InvalidSpec);
        }
        let transaction = self
            .connection
            .transaction_with_behavior(TransactionBehavior::Immediate)?;
        let (spec_json, spec_hash, spec_revision, policy_revision, generation): (String,String,i64,i64,i64) = transaction.query_row(
            "SELECT s.spec_json,s.spec_hash,s.spec_revision,s.policy_revision,t.generation FROM task_specs s JOIN tasks t USING(task_id)
             WHERE s.task_id=?1 AND t.recorded_state='running'", [task_id],
             |r| Ok((r.get(0)?,r.get(1)?,r.get(2)?,r.get(3)?,r.get(4)?)))
            .optional()?.ok_or(StoreError::Conflict)?;
        if digest(&spec_json) != spec_hash {
            return Err(StoreError::InvalidSpec);
        }
        let spec: TaskSpec =
            serde_json::from_str(&spec_json).map_err(|_| StoreError::InvalidSpec)?;
        let (canonical, path, allowed) = match tool_name {
            "read_input" => {
                let parsed: ReadArgs =
                    serde_json::from_str(args).map_err(|_| StoreError::InvalidSpec)?;
                (
                    serde_json::to_string(&parsed),
                    spec.task.input,
                    spec.task.allow_read,
                )
            }
            "write_output" => {
                let parsed: WriteArgs =
                    serde_json::from_str(args).map_err(|_| StoreError::InvalidSpec)?;
                if parsed.content.len() as u64 > FILE_LIMIT {
                    return Err(StoreError::InvalidSpec);
                }
                (
                    serde_json::to_string(&parsed),
                    spec.task.output,
                    spec.task.allow_write,
                )
            }
            _ => return Err(StoreError::InvalidSpec),
        };
        let args_json = canonical.map_err(|_| StoreError::InvalidSpec)?;
        let scope = path
            .to_str()
            .filter(|_| path.is_absolute())
            .ok_or(StoreError::InvalidSpec)?;
        let calls: i64 = transaction.query_row(
            "SELECT count(*) FROM tool_intents WHERE task_id=?1",
            [task_id],
            |r| r.get(0),
        )?;
        if calls >= TOOL_CALL_LIMIT {
            return Err(StoreError::InvalidSpec);
        }
        let status = if !allowed {
            "denied"
        } else if spec.approval_mode == ApprovalMode::Manual {
            "pending"
        } else {
            "approved"
        };
        let intent_id = Uuid::new_v4().to_string();
        let approval_id = Uuid::new_v4().to_string();
        let args_hash = digest(&args_json);
        transaction.execute("INSERT INTO tool_intents(intent_id,task_id,generation,call_id,tool_name,tool_version,args_json,args_hash,scope_path,status) VALUES (?1,?2,?3,?4,?5,1,?6,?7,?8,?9)",
            params![intent_id,task_id,generation,call_id,tool_name,args_json,args_hash,scope,if allowed { "prepared" } else { "denied" }])?;
        transaction.execute("INSERT INTO approvals(approval_id,intent_id,spec_revision,policy_revision,spec_hash,args_hash,tool_version,status,decision_source,expires_at,uses_remaining,decided_at)
            VALUES (?1,?2,?3,?4,?5,?6,1,?7,?8,unixepoch()+?9,1,CASE WHEN ?7='pending' THEN NULL ELSE strftime('%Y-%m-%dT%H:%M:%fZ','now') END)",
            params![approval_id,intent_id,spec_revision,policy_revision,spec_hash,args_hash,status,if status == "pending" { "pending" } else { "task_policy" },APPROVAL_TTL_SECONDS])?;
        append(
            &transaction,
            task_id,
            if status == "pending" {
                "approval_requested"
            } else if allowed {
                "approval_granted"
            } else {
                "approval_denied"
            },
            Some(if status == "pending" {
                "waiting_approval"
            } else {
                "running"
            }),
            &approval_id,
        )?;
        transaction.commit()?;
        Ok(PreparedIntent {
            approval_id,
            task_id: task_id.into(),
            call_id: call_id.into(),
            args_hash,
            spec_revision,
        })
    }

    pub fn approvals(&self, task_id: &str) -> Result<Vec<ApprovalRecord>, StoreError> {
        self.status(task_id)?;
        let mut query = self.connection.prepare(
            "SELECT a.approval_id,i.task_id,a.spec_revision,a.policy_revision,i.tool_name,i.tool_version,i.scope_path,i.args_json,a.args_hash,a.status,a.uses_remaining,a.expires_at,a.decision_source
             FROM approvals a JOIN tool_intents i USING(intent_id) WHERE i.task_id=?1 ORDER BY i.rowid LIMIT 100")?;
        let rows = query.query_map([task_id], |r| {
            Ok(ApprovalRecord {
                approval_id: r.get(0)?,
                task_id: r.get(1)?,
                spec_revision: r.get(2)?,
                policy_revision: r.get(3)?,
                tool_name: r.get(4)?,
                tool_version: r.get(5)?,
                scope_path: r.get(6)?,
                args_json: r.get(7)?,
                args_hash: r.get(8)?,
                status: r.get(9)?,
                uses_remaining: r.get(10)?,
                expires_at: r.get(11)?,
                decision_source: r.get(12)?,
            })
        })?;
        Ok(rows.collect::<Result<Vec<_>, _>>()?)
    }

    pub fn decide(
        &mut self,
        approval_id: &str,
        spec_revision: i64,
        approve: bool,
    ) -> Result<(), StoreError> {
        let transaction = self
            .connection
            .transaction_with_behavior(TransactionBehavior::Immediate)?;
        let task_id: String = transaction.query_row(
            "SELECT i.task_id FROM approvals a JOIN tool_intents i USING(intent_id) JOIN tasks t USING(task_id) JOIN task_specs s USING(task_id)
             WHERE a.approval_id=?1 AND a.spec_revision=?2 AND a.spec_revision=s.spec_revision AND a.policy_revision=s.policy_revision AND a.spec_hash=s.spec_hash
             AND a.uses_remaining=1 AND a.expires_at>unixepoch() AND (a.status='pending' OR (?3=0 AND a.status='approved'))
             AND i.status='prepared' AND i.generation=t.generation AND t.recorded_state IN ('running','waiting_approval')",
            params![approval_id,spec_revision,approve], |r| r.get(0)).optional()?.ok_or(StoreError::Conflict)?;
        transaction.execute("UPDATE approvals SET status=?2,decision_source='local_cli',decided_at=strftime('%Y-%m-%dT%H:%M:%fZ','now') WHERE approval_id=?1", params![approval_id, if approve { "approved" } else { "denied" }])?;
        append(
            &transaction,
            &task_id,
            if approve {
                "approval_granted"
            } else {
                "approval_denied"
            },
            None,
            approval_id,
        )?;
        transaction.commit()?;
        Ok(())
    }

    pub fn claim(&mut self, prepared: &PreparedIntent) -> Result<Claim, StoreError> {
        let transaction = self
            .connection
            .transaction_with_behavior(TransactionBehavior::Immediate)?;
        let value: Option<(String,i64,bool)> = transaction.query_row(
            "SELECT a.status,a.uses_remaining,a.expires_at>unixepoch() FROM approvals a JOIN tool_intents i USING(intent_id) JOIN tasks t USING(task_id) JOIN task_specs s USING(task_id)
             WHERE a.approval_id=?1 AND i.task_id=?2 AND i.call_id=?3 AND a.args_hash=?4 AND i.args_hash=a.args_hash AND a.spec_revision=?5
             AND a.spec_revision=s.spec_revision AND a.policy_revision=s.policy_revision AND a.spec_hash=s.spec_hash AND a.tool_version=i.tool_version
             AND i.generation=t.generation AND t.recorded_state IN ('running','waiting_approval')",
            params![prepared.approval_id,prepared.task_id,prepared.call_id,prepared.args_hash,prepared.spec_revision], |r| Ok((r.get(0)?,r.get(1)?,r.get(2)?))).optional()?;
        let (status, uses, valid) = value.ok_or(StoreError::Conflict)?;
        if !valid || uses != 1 || status == "denied" || status == "revoked" {
            return Ok(Claim::Denied);
        }
        if status == "pending" {
            return Ok(Claim::Pending);
        }
        if status != "approved" {
            return Err(StoreError::Conflict);
        }
        transaction.execute(
            "UPDATE approvals SET uses_remaining=0 WHERE approval_id=?1",
            [&prepared.approval_id],
        )?;
        let changed = transaction.execute("UPDATE tool_intents SET status='dispatched' WHERE task_id=?1 AND call_id=?2 AND status='prepared'", params![prepared.task_id,prepared.call_id])?;
        if changed != 1 {
            return Err(StoreError::Conflict);
        }
        append(
            &transaction,
            &prepared.task_id,
            "tool_dispatched",
            Some("running"),
            &prepared.approval_id,
        )?;
        transaction.commit()?;
        Ok(Claim::Granted)
    }

    pub fn receipt(
        &mut self,
        task_id: &str,
        call_id: &str,
        succeeded: bool,
    ) -> Result<(), StoreError> {
        let transaction = self
            .connection
            .transaction_with_behavior(TransactionBehavior::Immediate)?;
        let intent_id: String = transaction.query_row("UPDATE tool_intents SET status=?3 WHERE task_id=?1 AND call_id=?2 AND status='dispatched' RETURNING intent_id",
            params![task_id,call_id,if succeeded { "succeeded" } else { "failed" }], |r| r.get(0)).optional()?.ok_or(StoreError::Conflict)?;
        append(
            &transaction,
            task_id,
            if succeeded {
                "tool_succeeded"
            } else {
                "tool_failed"
            },
            None,
            &intent_id,
        )?;
        transaction.commit()?;
        Ok(())
    }
}

const APPLICATION_ID: i64 = 0x41504754;
const SCHEMA_VERSION: i64 = 2;
pub const EVENT_PAGE_SIZE: i64 = 100;

#[derive(Debug, thiserror::Error)]
pub enum StoreError {

pub struct TaskStore {

    schema_version: i64,

impl TaskStore {
    pub fn open(path: &Path) -> Result<Self, StoreError> {
        Self::connect(path, true)
    }

    pub fn read(path: &Path) -> Result<Self, StoreError> {
        Self::connect(path, false)
    }

    fn connect(path: &Path, create: bool) -> Result<Self, StoreError> {
        if rusqlite::version_number() < 3_051_003 {
            return Err(StoreError::Incompatible);
        }
        let flags = if create {
            OpenFlags::SQLITE_OPEN_READ_WRITE | OpenFlags::SQLITE_OPEN_CREATE
        } else {
            OpenFlags::SQLITE_OPEN_READ_ONLY
        };
        let mut connection = Connection::open_with_flags(
            path,
            flags | OpenFlags::SQLITE_OPEN_NO_MUTEX | OpenFlags::SQLITE_OPEN_NOFOLLOW,
        )?;
        connection.busy_timeout(Duration::from_millis(500))?;
        let version: i64 = connection.pragma_query_value(None, "user_version", |r| r.get(0))?;
        let application: i64 =
            connection.pragma_query_value(None, "application_id", |r| r.get(0))?;
        if !matches!(
            (application, version),
            (0, 0) | (APPLICATION_ID, 1..=SCHEMA_VERSION)
        ) {
            return Err(StoreError::Incompatible);
        }
        if version == 0 && !create {
            return Err(StoreError::Incompatible);
        }
        if create && version < SCHEMA_VERSION {
            connection.pragma_update(None, "foreign_keys", false)?;
            let transaction =
                connection.transaction_with_behavior(TransactionBehavior::Immediate)?;
            let mut current_version: i64 =
                transaction.pragma_query_value(None, "user_version", |r| r.get(0))?;
            let current_application: i64 =
                transaction.pragma_query_value(None, "application_id", |r| r.get(0))?;
            if (current_application, current_version) == (0, 0) {
                let tables: i64 = transaction.query_row(
                    "SELECT count(*) FROM sqlite_schema WHERE name NOT LIKE 'sqlite_%'",
                    [],
                    |r| r.get(0),
                )?;
                if tables != 0 {
                    return Err(StoreError::Incompatible);
                }
                transaction.execute_batch(include_str!("store_v1.sql"))?;
                transaction.pragma_update(None, "application_id", APPLICATION_ID)?;
                current_version = 1;
            } else if current_application != APPLICATION_ID
                || !(1..=SCHEMA_VERSION).contains(&current_version)
            {
                return Err(StoreError::Incompatible);
            }
            if current_version == 1 {
                transaction.execute_batch(include_str!("store_v2.sql"))?;
            }
            transaction.pragma_update(None, "user_version", SCHEMA_VERSION)?;
            transaction.commit()?;
        }
        let journal: String = if create {
            connection.pragma_update_and_check(None, "journal_mode", "wal", |r| r.get(0))?
        } else {
            connection.pragma_query_value(None, "journal_mode", |r| r.get(0))?
        };
        connection.pragma_update(None, "synchronous", "FULL")?;
        connection.pragma_update(None, "foreign_keys", true)?;
        connection.pragma_update(None, "trusted_schema", false)?;
        let synchronous: i64 = connection.pragma_query_value(None, "synchronous", |r| r.get(0))?;
        let foreign_keys: i64 =
            connection.pragma_query_value(None, "foreign_keys", |r| r.get(0))?;
        if (create && journal != "wal") || synchronous != 2 || foreign_keys != 1 {
            return Err(StoreError::Incompatible);
        }
        let schema_version = connection.pragma_query_value(None, "user_version", |r| r.get(0))?;
        Ok(Self {
            connection,
            schema_version,
        })
    }

    pub fn start(&mut self) -> Result<(String, i64), StoreError> {
        self.create_task(None)
    }

    pub fn start_spec(&mut self, spec: &TaskSpec) -> Result<(String, i64), StoreError> {
        self.create_task(Some(spec))
    }

    fn create_task(&mut self, spec: Option<&TaskSpec>) -> Result<(String, i64), StoreError> {
        let spec_json = spec
            .map(serde_json::to_string)
            .transpose()
            .map_err(|_| StoreError::InvalidSpec)?;
        if spec_json.as_ref().is_some_and(|value| value.len() > 131072) {
            return Err(StoreError::InvalidSpec);
        }
        let task_id = Uuid::new_v4().to_string();
        let transaction = self
            .connection
            .transaction_with_behavior(TransactionBehavior::Immediate)?;
        let seq = transaction.query_row(
            "INSERT INTO tasks (task_id, recorded_state, last_seq) VALUES (?1, 'running', 1) RETURNING last_seq",
            [&task_id],
            |r| r.get::<_, i64>(0),
        )?;
        transaction.execute(
            "INSERT INTO events (event_id, task_id, seq, event_type) VALUES (?1, ?2, ?3, 'started')",
            params![Uuid::new_v4().to_string(), task_id, seq],
        )?;
        if let Some(spec_json) = spec_json {
            transaction.execute(
                "INSERT INTO task_specs(task_id,spec_json,spec_hash) VALUES (?1,?2,?3)",
                params![task_id, spec_json, approvals::digest(&spec_json)],
            )?;
        }
        transaction.commit()?;
        Ok((task_id, seq))
    }

    pub fn finish(
        &mut self,
        task_id: &str,
        expected_seq: i64,
        result: &Result<TaskReport, TaskError>,
    ) -> Result<(), StoreError> {
        let state = match result {
            Ok(report) => match report.artifact {
                ArtifactState::Verified => "succeeded",
                ArtifactState::AwaitingReview => "needs_review",
            },
            Err(TaskError::Cancelled { .. }) => "cancelled",
            Err(TaskError::Timeout | TaskError::Harness(_) | TaskError::State) => "outcome_unknown",
            Err(_) => "failed",
        };
        let transaction = self
            .connection
            .transaction_with_behavior(TransactionBehavior::Immediate)?;
        let artifact_created = match result {
            Ok(_) | Err(TaskError::IncompleteArtifact | TaskError::Verification(_)) => Some(true),
            Err(TaskError::Cancelled { artifact_created }) => Some(*artifact_created),
            Err(
                TaskError::InvalidTask
                | TaskError::InvalidInput
                | TaskError::Input(_)
                | TaskError::MissingArtifact,
            ) => Some(false),
            Err(_) => None,
        };
        let seq: Option<i64> = transaction.query_row(
            "UPDATE tasks SET recorded_state = ?1, last_seq = last_seq + 1, artifact_created = ?4,
             updated_at = strftime('%Y-%m-%dT%H:%M:%fZ', 'now')
             WHERE task_id = ?2 AND recorded_state IN ('running','waiting_approval') AND last_seq = ?3 RETURNING last_seq",
            params![state, task_id, expected_seq, artifact_created], |r| r.get(0),
        ).optional()?;
        let seq = match seq {
            Some(seq) => seq,
            None => {
                let exists: bool = transaction.query_row(
                    "SELECT EXISTS(SELECT 1 FROM tasks WHERE task_id=?1)",
                    [task_id],
                    |r| r.get(0),
                )?;
                return Err(if exists {
                    StoreError::Conflict
                } else {
                    StoreError::NotFound
                });
            }
        };
        transaction.execute(
            "INSERT INTO events (event_id, task_id, seq, event_type) VALUES (?1, ?2, ?3, ?4)",
            params![Uuid::new_v4().to_string(), task_id, seq, state],
        )?;
        transaction.execute("UPDATE approvals SET status='revoked' WHERE uses_remaining=1 AND status IN ('pending','approved') AND intent_id IN (SELECT intent_id FROM tool_intents WHERE task_id=?1)", [task_id])?;
        transaction.execute(
            "UPDATE tool_intents SET status='cancelled' WHERE task_id=?1 AND status='prepared'",
            [task_id],
        )?;
        transaction.commit()?;
        Ok(())
    }

    pub fn status(&self, task_id: &str) -> Result<TaskRecord, StoreError> {
        self.connection
            .query_row(
                "SELECT task_id, schema_version, generation, recorded_state, last_seq, updated_at, artifact_created
             FROM tasks WHERE task_id = ?1",
                [task_id],
                |r| {
                    Ok(TaskRecord {
                        task_id: r.get(0)?,
                        schema_version: r.get(1)?,
                        generation: r.get(2)?,
                        recorded_state: r.get(3)?,
                        last_seq: r.get(4)?,
                        updated_at: r.get(5)?,
                        artifact_created: r.get(6)?,
                    })
                },
            )
            .optional()?
            .ok_or(StoreError::NotFound)
    }

    pub fn events(&self, task_id: &str, after_seq: i64) -> Result<Vec<TaskEvent>, StoreError> {
        if after_seq < 0 {
            return Err(StoreError::InvalidCursor);
        }
        self.status(task_id)?;
        let related = if self.schema_version == 1 {
            "NULL"
        } else {
            "related_id"
        };
        let mut query = self.connection.prepare(&format!(
            "SELECT event_id, task_id, schema_version, generation, seq, event_type, created_at, {related}
             FROM events WHERE task_id = ?1 AND seq > ?2 ORDER BY seq LIMIT ?3"
        ))?;
        let rows = query.query_map(params![task_id, after_seq, EVENT_PAGE_SIZE], |r| {
            Ok(TaskEvent {
                event_id: r.get(0)?,
                task_id: r.get(1)?,
                schema_version: r.get(2)?,
                generation: r.get(3)?,
                seq: r.get(4)?,
                event_type: r.get(5)?,
                created_at: r.get(6)?,
                related_id: r.get(7)?,
            })
        })?;
        Ok(rows.collect::<Result<Vec<_>, _>>()?)
    }
}

fn open_store(create: bool) -> Result<TaskStore, String> {
    let directory = match std::env::var_os("AGENT_STATE_DIR") {
        Some(path) if !path.is_empty() => PathBuf::from(path),
        Some(_) => return Err("AGENT_STATE_DIR 不可為空".into()),
        None => directories::ProjectDirs::from("", "", "agent-platform")
            .map(|dirs| dirs.data_local_dir().to_path_buf())
            .ok_or("找不到使用者資料目錄，請設定 AGENT_STATE_DIR")?,
    };
    if !directory.is_absolute() {
        return Err("AGENT_STATE_DIR 必須是絕對路徑".into());
    }
    if create {
        let mut builder = std::fs::DirBuilder::new();
        builder.recursive(true);
        #[cfg(unix)]
        {
            use std::os::unix::fs::DirBuilderExt;
            builder.mode(0o700);
        }
        builder
            .create(&directory)
            .map_err(|_| "無法建立任務資料目錄")?;
    }
    let metadata = std::fs::symlink_metadata(&directory).map_err(|_| "找不到任務資料目錄")?;
    if !metadata.is_dir() {
        return Err("任務資料目錄必須是一般目錄，不可使用符號連結".into());
    }
    #[cfg(unix)]
    {
        use std::os::unix::fs::PermissionsExt;
        if metadata.permissions().mode() & 0o077 != 0 {
            return Err("任務資料目錄不可讓其他使用者存取，請設定權限為 700".into());
        }
    }
    let path = directory.join("tasks.sqlite3");
    match std::fs::symlink_metadata(&path) {
        Ok(metadata) => {
            if !metadata.is_file() {
                return Err("任務資料庫必須是一般檔案，不可使用符號連結".into());
            }
            #[cfg(unix)]
            {
                use std::os::unix::fs::MetadataExt;
                if metadata.nlink() != 1 {
                    return Err("任務資料庫不可使用硬連結".into());
                }
            }
        }
        Err(error) if create && error.kind() == std::io::ErrorKind::NotFound => {}
        Err(_) => return Err("找不到或無法讀取任務資料庫".into()),
    }
    if create {
        TaskStore::open(&path)
    } else {
        TaskStore::read(&path)
    }
    .map_err(|error| error.to_string())
}

async fn run() -> Result<ExitCode, String> {
    let args: Vec<_> = std::env::args_os().skip(1).collect();
    if args.len() == 1 && args[0] == "--help" {
        println!(
            "用法：agent-platform run <任務.json>\n      agent-platform status <任務編號>\n      agent-platform events <任務編號> [已讀序號]\nevents 每次最多 100 筆，下一頁使用最後一筆的 seq 作已讀序號。\n目前為開發入口，只連本機模型服務；讀寫須在任務中明確授權。尚非企業正式發布。"
        );
        return Ok(ExitCode::SUCCESS);
    }
    if args.len() == 2 && args[0] == "status" {
        let task_id = args[1].to_str().ok_or("任務編號格式錯誤")?;
        let record = open_store(false)?
            .status(task_id)
            .map_err(|error| error.to_string())?;
        if record.recorded_state == "running" {
            eprintln!("這是最後保存的狀態，不能確認程序仍在執行；請先核對成果，不要直接重跑。");
        }
        println!(
            "{}",
            serde_json::to_string(&record).map_err(|_| "無法輸出任務狀態")?
        );
        return Ok(ExitCode::SUCCESS);
    }
    if args.len() == 2 && args[0] == "approvals" {
        let task_id = args[1].to_str().ok_or("任務編號格式錯誤")?;
        let approvals = open_store(false)?
            .approvals(task_id)
            .map_err(|error| error.to_string())?;
        println!(
            "{}",
            serde_json::to_string(&approvals).map_err(|_| "無法輸出批准紀錄")?
        );
        return Ok(ExitCode::SUCCESS);
    }
    if args.len() == 3 && (args[0] == "approve" || args[0] == "deny") {
        let approval_id = args[1].to_str().ok_or("批准編號格式錯誤")?;
        let revision = args[2]
            .to_str()
            .and_then(|value| value.parse::<i64>().ok())
            .filter(|value| *value > 0)
            .ok_or("必須提供正整數的任務版本")?;
        let approve = args[0] == "approve";
        open_store(true)?
            .decide(approval_id, revision, approve)
            .map_err(|error| error.to_string())?;
        println!(
            "{}",
            serde_json::json!({"approval_id":approval_id,"status":if approve { "approved" } else { "denied" }})
        );
        return Ok(ExitCode::SUCCESS);
    }
    if (args.len() == 2 || args.len() == 3) && args[0] == "events" {
        let task_id = args[1].to_str().ok_or("任務編號格式錯誤")?;
        let after_seq = match args.get(2) {
            Some(value) => value
                .to_str()
                .and_then(|value| value.parse::<i64>().ok())
                .ok_or("事件位置格式錯誤")?,
            None => 0,
        };
        for event in open_store(false)?
            .events(task_id, after_seq)
            .map_err(|error| error.to_string())?
        {
            println!(
                "{}",
                serde_json::to_string(&event).map_err(|_| "無法輸出任務事件")?
            );
        }
        return Ok(ExitCode::SUCCESS);
    }
    if args.len() != 2 || args[0] != "run" {
        return Err("用法：agent-platform run <任務.json>".into());
    }
    let mut bytes = Vec::new();
    agent_platform_agent::input::open_checked(std::path::Path::new(&args[1]), u64::MAX)
        .map_err(|_| "無法開啟任務檔")?
        .take(65537)
        .read_to_end(&mut bytes)
        .map_err(|_| "無法讀取任務檔")?;
    if bytes.len() > 65536 {
        return Err("任務檔超過 64 KiB".into());
    }
    let mut request: Request =
        serde_json::from_slice(&bytes).map_err(|_| "任務 JSON 格式錯誤或包含未知欄位")?;
    let loopback = match request.endpoint.host() {
        Some(url::Host::Ipv4(ip)) => ip.is_loopback(),
        Some(url::Host::Ipv6(ip)) => ip.is_loopback(),
        _ => false,
    };
    if !loopback
        || !matches!(request.endpoint.scheme(), "http" | "https")
        || !request.endpoint.username().is_empty()
        || request.endpoint.password().is_some()
        || request.endpoint.query().is_some()
        || request.endpoint.fragment().is_some()
        || request.model.trim().is_empty()
    {
        return Err(
            "目前只接受明確的本機 IP 模型入口與非空模型名稱；網址不得夾帶憑證或查詢參數".into(),
        );
    }
    let key = match std::env::var("AGENT_API_KEY") {
        Ok(value) => value,
        Err(std::env::VarError::NotPresent) => String::new(),
        Err(_) => return Err("AGENT_API_KEY 格式錯誤".into()),
    };
    let http = reqwest::Client::builder()
        .no_proxy()
        .redirect(reqwest::redirect::Policy::none())
        .timeout(Duration::from_secs(60))
        .build()
        .map_err(|_| "無法建立模型連線")?;
    let client = OpenAIConfig::new(key)
        .with_base_url(request.endpoint.as_str())
        .connect(ReqwestClient::from(http));
    let cancellation = CancellationToken::new();
    let signal_cancellation = cancellation.clone();
    let mut interrupted = false;
    ctrlc::try_set_handler(move || {
        if interrupted {
            std::process::exit(130);
        }
        interrupted = true;
        signal_cancellation.cancel();
    })
    .map_err(|_| "無法接收停止要求，任務未啟動")?;
    let workspace = std::env::current_dir().map_err(|_| "無法取得工作目錄")?;
    request.task.input = workspace.join(&request.task.input);
    request.task.output = workspace.join(&request.task.output);
    let spec = TaskSpec {
        workspace,
        endpoint: request.endpoint.to_string(),
        model: request.model.clone(),
        task: request.task.clone(),
        approval_mode: request.approval_mode,
    };
    let mut store = open_store(true)?;
    let (task_id, _) = store.start_spec(&spec).map_err(|error| error.to_string())?;
    let store = Arc::new(Mutex::new(store));
    let gate = ApprovalGate::new(
        store.clone(),
        task_id.clone(),
        cancellation.clone(),
        request.approval_mode == ApprovalMode::Manual,
    );
    eprintln!("任務編號：{task_id}");
    if request.approval_mode == ApprovalMode::Manual {
        eprintln!(
            "逐次確認模式：請以 approvals 查閱這個任務，再用 approve 或 deny 提交決定及任務版本。"
        );
    }
    let mut result = run_file_task(
        client.chat(request.model),
        request.task,
        cancellation,
        gate.clone(),
    )
    .await;
    if matches!(result, Err(TaskError::Harness(_))) && gate.was_denied() {
        result = Err(TaskError::ApprovalDenied);
    }
    let mut store = store.lock().map_err(|_| "任務紀錄狀態異常")?;
    let seq = store
        .status(&task_id)
        .map_err(|error| error.to_string())?
        .last_seq;
    store
        .finish(&task_id, seq, &result)
        .map_err(|error| format!("{error}；任務結果未確認，請勿直接重跑"))?;
    let report = match result {
        Err(TaskError::Cancelled { artifact_created }) => {
            println!(
                "{}",
                serde_json::json!({"task_id": task_id, "state": "cancelled", "artifact_created": artifact_created})
            );
            eprintln!("已停止本機任務；已建立的成果會保留，請核對後再接續。");
            return Ok(ExitCode::from(130));
        }
        other => other.map_err(|error| error.to_string())?,
    };
    println!(
        "{}",
        serde_json::json!({"task_id": task_id, "artifact": report.artifact, "assistant_message": report.assistant_message})
    );
    Ok(if report.artifact == ArtifactState::Verified {
        ExitCode::SUCCESS
    } else {
        ExitCode::from(2)
    })
}

#[tokio::main(flavor = "current_thread")]
async fn main() -> ExitCode {
    match run().await {
        Ok(code) => code,
        Err(error) => {
            eprintln!("{error}");
            ExitCode::FAILURE
        }
    }
}

impl Tool for ReadInput {
    const NAME: &'static str = "read_input";
    type Args = ReadArgs;
    type Output = String;
    type Error = ToolExecutionError;

    fn description(&self) -> String {
        "讀取使用者指定且獲准的來源檔案".into()
    }
    fn parameters(&self) -> serde_json::Value {
        json!({"type":"object","properties":{},"additionalProperties":false})
    }
    async fn call(&self, _: &mut ToolContext, _: ReadArgs) -> Result<String, Self::Error> {
        if self.1.is_cancelled() {
            return Err(ToolExecutionError::refused("任務已要求停止，不再讀取"));
        }
        let mut guard = self
            .0
            .lock()
            .map_err(|_| ToolExecutionError::refused("來源狀態異常"))?;
        let file = guard
            .as_mut()
            .ok_or_else(|| ToolExecutionError::refused("使用者未授權讀取來源"))?;
        file.seek(SeekFrom::Start(0))
            .map_err(ToolExecutionError::from_error)?;
        let mut content = String::new();
        file.take(FILE_LIMIT + 1)
            .read_to_string(&mut content)
            .map_err(ToolExecutionError::from_error)?;
        if content.len() as u64 > FILE_LIMIT {
            return Err(ToolExecutionError::refused("來源超出大小限制"));
        }
        Ok(content)
    }
}

impl Tool for WriteOutput {
    const NAME: &'static str = "write_output";
    type Args = WriteArgs;
    type Output = String;
    type Error = ToolExecutionError;

    fn description(&self) -> String {
        "建立使用者指定且獲准的新成果檔案，不覆寫既有資料".into()
    }
    fn parameters(&self) -> serde_json::Value {
        json!({"type":"object","properties":{"content":{"type":"string"}},"required":["content"],"additionalProperties":false})
    }
    async fn call(&self, _: &mut ToolContext, args: WriteArgs) -> Result<String, Self::Error> {
        if self.cancellation.is_cancelled() {
            return Err(ToolExecutionError::refused("任務已要求停止，不再寫入"));
        }
        if !self.allowed {
            return Err(ToolExecutionError::refused("使用者未授權寫入成果"));
        }
        if args.content.len() as u64 > FILE_LIMIT {
            return Err(ToolExecutionError::refused("成果超出大小限制"));
        }
        let mut artifact = self
            .artifact
            .lock()
            .map_err(|_| ToolExecutionError::refused("成果狀態異常"))?;
        if artifact.is_some() {
            return Err(ToolExecutionError::refused("成果已寫入，不重複執行"));
        }
        let file = OpenOptions::new()
            .read(true)
            .write(true)
            .create_new(true)
            .open(&self.path)
            .map_err(ToolExecutionError::from_error)?;
        let artifact = artifact.insert(Artifact {
            file,
            complete: false,
        });
        artifact
            .file
            .write_all(args.content.as_bytes())
            .map_err(ToolExecutionError::from_error)?;
        artifact
            .file
            .sync_all()
            .map_err(ToolExecutionError::from_error)?;
        artifact.complete = true;
        Ok("成果已寫入，仍須獨立驗證".into())
    }
}

pub async fn run_file_task(
    model: impl Into<rig_core::DynModel<rig_core::operation::Completion>>,
    task: FileTask,
    cancellation: CancellationToken,
    hook: impl rig_agent::agent::AgentHook + 'static,
) -> Result<TaskReport, TaskError> {
    if cancellation.is_cancelled() {
        return Err(TaskError::Cancelled {
            artifact_created: false,
        });
    }
    if task.prompt.trim().is_empty()
        || task.prompt.len() as u64 > FILE_LIMIT
        || task
            .expected
            .as_ref()
            .is_some_and(|s| s.len() as u64 > FILE_LIMIT)
    {
        return Err(TaskError::InvalidTask);
    }
    let input = if task.allow_read {
        let file =
            crate::input::open_checked(&task.input, FILE_LIMIT).map_err(|error| match error {
                crate::input::InputError::InvalidFile => TaskError::InvalidInput,
                crate::input::InputError::Io(error) => TaskError::Input(error),
            })?;
        Some(file)
    } else {
        None
    };
    let artifact = Arc::new(Mutex::new(None));
    let agent = AgentBuilder::new(model)
        .add_hook(hook)
        .tool(ReadInput(Arc::new(Mutex::new(input)), cancellation.clone()))
        .tool(WriteOutput {
            path: task.output,
            allowed: task.allow_write,
            artifact: artifact.clone(),
            cancellation: cancellation.clone(),
        })
        .build();
    let response = tokio::select! {
        biased;
        _ = cancellation.cancelled() => {
            let artifact_created = artifact.lock().map_err(|_| TaskError::State)?.is_some();
            return Err(TaskError::Cancelled { artifact_created });
        }
        response = tokio::time::timeout(TASK_TIMEOUT, agent.prompt(task.prompt).max_turns(8).tool_concurrency(1).run()) => {
            response.map_err(|_| TaskError::Timeout)?
                .map_err(|error| TaskError::Harness(Box::new(error)))?
        }
    };
    let mut guard = artifact.lock().map_err(|_| TaskError::State)?;
    if cancellation.is_cancelled() {
        return Err(TaskError::Cancelled {
            artifact_created: guard.is_some(),
        });
    }
    let artifact = guard.as_mut().ok_or(TaskError::MissingArtifact)?;
    if !artifact.complete {
        return Err(TaskError::IncompleteArtifact);
    }
    let state = if let Some(expected) = task.expected {
        verify_file(&mut artifact.file, expected.as_bytes()).map_err(TaskError::Verification)?;
        ArtifactState::Verified
    } else {
        ArtifactState::AwaitingReview
    };
    if cancellation.is_cancelled() {
        return Err(TaskError::Cancelled {
            artifact_created: true,
        });
    }
    Ok(TaskReport {
        artifact: state,
        assistant_message: response.output().to_owned(),
    })
}

#[cfg(test)]
mod tests {
    use super::*;

    #[tokio::test]
    async fn cancelled_tools_refuse_before_touching_files() -> Result<(), Box<dyn std::error::Error>>
    {
        let cancellation = CancellationToken::new();
        cancellation.cancel();
        let reader = ReadInput(Arc::new(Mutex::new(None)), cancellation.clone());
        let error = reader
            .call(&mut ToolContext::default(), ReadArgs {})
            .await
            .err();
        assert!(error.is_some_and(|error| error.to_string().contains("不再讀取")));
        let stamp = std::time::SystemTime::now()
            .duration_since(std::time::UNIX_EPOCH)?
            .as_nanos();
        let path = std::env::temp_dir().join(format!(
            "agent-cancelled-tool-{}-{stamp}",
            std::process::id()
        ));
        let writer = WriteOutput {
            path: path.clone(),
            allowed: true,
            artifact: Arc::new(Mutex::new(None)),
            cancellation,
        };
        let error = writer
            .call(
                &mut ToolContext::default(),
                WriteArgs {
                    content: "must not write".into(),
                },
            )
            .await
            .err();
        assert!(error.is_some_and(|error| error.to_string().contains("不再寫入")));
        assert!(!path.exists());
        Ok(())
    }
}
